1/**
2 * @file include/types/attributes/attributes.h
3 * @brief Estruturas e enumerações que representam atributos da JVM presentes em fields, methods e classes.
4 *
5 * A JVM Specification define diversos atributos que podem aparecer em:
6 * - fields
7 * - methods
8 * - Code (dentro de um método)
9 * - ClassFile
10 *
11 * Cada atributo é identificado por um nome (UTF8) no constant pool e possui
12 * um formato específico. Este arquivo define:
13 *
14 * 1. O enum `attribute_name`, que lista todos os atributos suportados.
15 * 2. Estruturas auxiliares usadas por atributos específicos.
16 * 3. A estrutura principal `attribute`, que contém o valor do atributo
17 * como uma união (`union`) de todos os tipos possíveis.
18 *
19 * O parser seleciona a variante correta do `union` com base no nome do atributo.
20 */
21

35/**
36 * @enum attribute_name
37 * @brief Enumeração para identificação dos tipos de atributos suportados.
38 *
39 * O parser converte o nome UTF8 do atributo (obtido do constant pool)
40 * para um dos valores desta enum via `convert_attr_name()`.
41 *
42 * Exemplos de atributos definidos pela JVM:
43 * - Code
44 * - LineNumberTable
45 * - LocalVariableTable
46 * - Exceptions
47 * - Signature
48 * - RuntimeVisibleAnnotations
49 */
50typedef enum attribute_name
51{
52 ConstantValue,
53 Code,
54 StackMapTable,
55 Exceptions,
56 InnerClasses,
57 EnclosingMethod,
58 Synthetic,
59 Signature,
60 SourceFile,
61 SourceDebugExtension,
62 LineNumberTable,
63 LocalVariableTable,
64 LocalVariableTypeTable,
65 Deprecated,
66 RuntimeVisibleAnnotations,
67 RuntimeInvisibleAnnotations,
68 RuntimeVisibleParameterAnnotations,
69 RuntimeInvisibleParameterAnnotations,
70 AnnotationDefault,
71 BootstrapMethods,
72 MethodParameters
73} attribute_name;

104} exception_table;

159/**
160 * @struct attribute
161 * @brief Representa um atributo genérico da JVM.
162 *
163 * Cada atributo começa com:
164 * - attribute_name_index (u2)
165 * - attribute_length (u4)
166 *
167 * Depois, o formato interno depende do nome do atributo,
168 * armazenado aqui no `union info`.
169 */
170typedef struct attribute
171{
172 u2 attribute_name_index; /**< Índice no constant pool para o nome do atributo. */
173 u4 attribute_length; /**< Tamanho dos dados (não inclui os 6 bytes iniciais). */
174
175 /**
176 * @union info
177 * @brief Conteúdo do atributo, variando conforme seu tipo.
178 */
179 union
180 {

182 struct
183 {
184 u2 constantvalue_index;
185 } ConstantValue;

200 } Code;
