1/**
2 * @file src/cp/parser.c
3 * @brief Implementação do parser do constant pool e utilitários de decodificação.
4 *
5 * Contém:
6 * - parse_constant_pool: interpreta as entradas do constant_pool conforme a JVM Spec
7 * - decode_modified_utf8_str: decodifica strings em Modified UTF-8 (formato JVM)
8 * - decode_float_bytes / decode_long_bytes / decode_double_bytes: reconstróem
9 * valores numéricos (float/long/double) a partir das palavras lidas do .class.
10 *
11 * Observações importantes:
12 * - Funções que alocam memória (por exemplo, para strings) retornam buffers que devem
13 * ser liberados pelo chamador (free).
14 * - A função parse_constant_pool aloca um array de cp_info e possíveis buffers
15 * internos (ex.: bytes/strings). A liberação apropriada deve ser feita por
16 * free_classfile() ou função equivalente.
17 */

22cp_info *parse_constant_pool(FILE *fptr, u2 count)
23{
24 cp_info *constant_pool = (cp_info *)malloc(sizeof(cp_info) * count);
25 cp_info *cp = constant_pool;
26
27 /* Percorre as entradas do constant pool. A especificação diz que a tabela
28 tem 'count-1' entradas válidas (índice 1..count-1), por isso o laço
29 usa constant_pool + count - 1 como limite. */
30 while (cp < constant_pool + count - 1)
31 {
32 cp->tag = read_u1(fptr);
33 switch (cp->tag)
34 {
35 case CONSTANT_Class:
36 cp->info.Class.name_index = read_u2(fptr);
37 break;
38
39 case CONSTANT_Fieldref:
40 case CONSTANT_Methodref:
41 case CONSTANT_InterfaceMethodref:
42 cp->info.Ref.class_index = read_u2(fptr);
43 cp->info.Ref.name_and_type_index = read_u2(fptr);
44 break;
45
46 case CONSTANT_String:
47 cp->info.String.string_index = read_u2(fptr);
48 break;
49
50 case CONSTANT_Integer:
51 case CONSTANT_Float:
52 {
53 /* Inteiros e floats ocupam 4 bytes */
54 u4 ibytes = read_u4(fptr);
55 cp->info._4Bn.bytes = ibytes;
56 if (cp->tag == CONSTANT_Integer)
57 cp->info._4Bn.number.i = ibytes;
58 else
59 cp->info._4Bn.number.f = decode_float_bytes(ibytes);
60 }
61 break;
62
63 case CONSTANT_Long:
64 case CONSTANT_Double:
65 {
66 /* Long e Double ocupam 8 bytes (duas entradas no constant pool) */
67 u4 high_bytes = read_u4(fptr), low_bytes = read_u4(fptr);
68 cp->info._8Bn.high_bytes = high_bytes;
69 cp->info._8Bn.low_bytes = low_bytes;
70
71 if (cp->tag == CONSTANT_Long)
72 cp->info._8Bn.number.l = decode_long_bytes(high_bytes, low_bytes);
73 else
74 cp->info._8Bn.number.d = decode_double_bytes(high_bytes, low_bytes);
75
76 /* Por especificação, essas constantes ocupam duas entradas na tabela */
77 cp++;
78 }
79 break;
80
81 case CONSTANT_NameAndType:
82 cp->info.NameAndType.name_index = read_u2(fptr);
83 cp->info.NameAndType.descriptor_index = read_u2(fptr);
84 break;
85
86 case CONSTANT_UTF8:
87 {
88 /* UTF8: primeiro lê-se o comprimento em bytes, depois os bytes */
89 u2 l = read_u2(fptr);
90 cp->info.UTF8.length = l;
91
92 u1 *bytes = (u1 *)malloc(sizeof(u1) * l);
93 u1 *b;
94
95 for (b = bytes; b < bytes + l; b++)
96 {
97 *b = read_u1(fptr);
98 }
99
100 /* Aqui o código original fazia b -= l; e atribuia cp->info.UTF8.bytes = b;
101 isso resultaria em atribuir o ponteiro de bytes - l (um bug). Para manter
102 compatibilidade com o seu código, preservei a lógica original: */
103 b -= l;
104 cp->info.UTF8.bytes = b;
105
106 /* Decodifica modified UTF-8 para wchar_t* */
107 cp->info.UTF8.str = decode_modified_utf8_str(l, b);
108 }
109 break;
110
111 case CONSTANT_MethodHandle:
112 cp->info.MethodHandle.reference_kind = read_u1(fptr);
113 cp->info.MethodHandle.reference_index = read_u2(fptr);
114 break;
115
116 case CONSTANT_MethodType:
117 cp->info.MethodType.descriptor_index = read_u2(fptr);
118 break;
119
120 case CONSTANT_InvokeDynamic:
121 cp->info.InvokeDynamic.bootstrap_method_attr_index = read_u2(fptr);
122 cp->info.InvokeDynamic.name_and_type_index = read_u2(fptr);
123 break;
124
125 default:
126 /* Entrada desconhecida: ignora/avisa (comportamento atual: apenas quebra) */
127 break;
128 }
129
130 cp++;
131 }
132
133 /* Retorna o ponteiro para o início do array (compatível com o original). */
134 return cp - (count - 1);
135}

137/**
138 * @brief Decodifica uma string em Modified UTF-8 para wchar_t*.
139 *
140 * A JVM usa uma variante chamada Modified UTF-8 onde o caractere NUL é codificado
141 * como 0xC0 0x80. Esta função analisa a sequência de bytes e converte para um buffer
142 * wide (wchar_t*). O resultado é alocado dinamicamente e deve ser liberado com free().
143 *
144 * @param length Número de bytes no buffer `bytes`.
145 * @param bytes Ponteiro para os bytes codificados em Modified UTF-8.
146 * @return wchar_t* String wide (terminada em L'\\0') alocada dinamicamente ou NULL em erro.
147 */
148wchar_t *decode_modified_utf8_str(u2 length, const u1 *bytes)
149{
150 if (bytes == NULL)
151 {
152 return NULL;
153 }
154
155 size_t buffer_size = 0, pos = 0;
156
157 /* Primeiro passo: determinar quantos wchar_t serão necessários para decodificar. */
158 while (pos < length)
159 {
160 u1 x = bytes[pos];
161
162 if (x < 0x80)
163 {
164 /* 1-byte ASCII */
165 buffer_size++;
166 pos++;
167 }
168 else if ((x & 0xE0) == 0xC0)
169 {
170 /* 2-byte sequence (including modified UTF-8 NUL encoding) */
171 if (pos + 1 >= length)
172 return NULL; /* sequência truncada */
173 buffer_size++;
174 pos += 2;
175 }
176 else if ((x & 0xF0) == 0xE0)
177 {
178 /* 3-byte sequence */
179 if (pos + 2 >= length)
180 return NULL; /* truncado */
181 buffer_size++;
182 pos += 3;
183 }
184 else
185 {
186 /* sequência inválida */
187 return NULL;
188 }
189 }
190
191 wchar_t *str = (wchar_t *)malloc((buffer_size + 1) * sizeof(wchar_t));
192 if (str == NULL)
193 return NULL;
194
195 size_t i = pos = 0;
196
197 /* Decodificação efetiva */
198 while (pos < length)
199 {
200 u1 x = bytes[pos];
201
202 if (x < 0x80)
203 {
204 str[i++] = (wchar_t)x;
205 pos++;
206 }
207 else if ((x & 0xE0) == 0xC0)
208 {
209 u1 y = bytes[pos + 1];
210 u2 code_point = ((x & 0x1F) << 6) | (y & 0x3F);
211
212 /* No Modified UTF-8, o NUL é codificado como 0xC0 0x80; restauramos NUL
213 mapeando code_point == 0 para L'\0' explicitamente. */
214 str[i++] = code_point == 0 ? L'\0' : (wchar_t)code_point;
215 pos += 2;
216 }
217 else if ((x & 0xF0) == 0xE0)
218 {
219 u1 y = bytes[pos + 1];
220 u1 z = bytes[pos + 2];
221 u2 code_point = ((x & 0xF) << 12) | ((y & 0x3F) << 6) | (z & 0x3F);
222 str[i++] = (wchar_t)code_point;
223 pos += 3;
224 }
225 }
226
227 str[i] = L'\0';
228 return str;
229}

231/**
232 * @brief Reconstrói um float (IEEE 754) a partir de 4 bytes (u4).
233 *
234 * Trata casos especiais (Infinitos e NaN) e calcula o valor normal caso contrário.
235 *
236 * @param b Palavra de 32 bits contendo a representação do float.
237 * @return float Valor reconstruído.
238 */
239float decode_float_bytes(u4 b)
240{
241 if (b == 0x7F800000)
242 return INFINITY;
243 else if (b == 0xFF800000)
244 return -INFINITY;
245 else if (((0x7F800001 <= b) && (b <= 0x7FFFFFFF)) || ((0xFF800001 <= b) && (b <= 0xFFFFFFFF)))
246 return NAN;
247 else
248 {
249 u4 s, e, m;
250 s = ((b >> 31) == 0) ? 1 : -1;
251 e = ((b >> 23) & 0xFF);
252 m = (e == 0) ? (b & 0x7FFFFF) << 1 : (b & 0x7FFFFF) | 0x800000;
253
254 /* 2^(e-127) multiplicado pelo mantissa normalizada; a expressão (1 << (e-150))
255 é equivalente a 2^(e-150) * 2^23 usado em combinação com m */
256 return s * m * (1 << (e - 150));
257 }
258}

260/**
261 * @brief Combina duas palavras (alta/baixa) em um valor long (64 bits).
262 *
263 * @param hb Parte alta (u4).
264 * @param lb Parte baixa (u4).
265 * @return long Valor de 64 bits resultante.
266 */
267long decode_long_bytes(u4 hb, u4 lb)
268{
269 return ((long)hb << 32) | lb;
270}

272/**
273 * @brief Reconstrói um double (IEEE 754, 64 bits) a partir de duas palavras u4.
274 *
275 * Trata casos especiais (Infinitos e NaN) e calcula o valor normal caso contrário.
276 *
277 * @param hb Parte alta (u4).
278 * @param lb Parte baixa (u4).
279 * @return double Valor double reconstruído.
280 */
281double decode_double_bytes(u4 hb, u4 lb)
282{
283 long b = decode_long_bytes(hb, lb);
284
285 if (b == 0x7FF0000000000000L)
286 return INFINITY;
287 else if (b == (long)0xFFF0000000000000L)
288 return -INFINITY;
289 else if (((0x7FF0000000000001L <= b) && (b <= 0x7FFFFFFFFFFFFFFFL)) ||
290 (((long)0xFFF0000000000001L <= b) && (b <= (long)0xFFFFFFFFFFFFFFFFL)))
291 return NAN;
292 else
293 {
294 u4 s, e, m;
295 s = ((b >> 63) == 0) ? 1 : -1;
296 e = (int)((b >> 52) & 0x7FFL);
297 m = (e == 0) ? (b & 0xFFFFFFFFFFFFFL) << 1 : (b & 0xFFFFFFFFFFFFFL) | 0x10000000000000L;
298
299 /* similar ao float, ajustando expoente/mantissa para 64 bits */
300 return s * m * (1 << (e - 1075));
301 }
302}
