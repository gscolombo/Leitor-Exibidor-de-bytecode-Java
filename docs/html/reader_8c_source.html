101 }
102

110/**
111 * @brief Lê um arquivo .class inteiro e popula uma estrutura ClassFile.
112 *
113 * A função procura o tamanho do arquivo, imprime-o, reposiciona o ponteiro e
114 * realiza a leitura sequencial dos componentes do classfile:
115 * - magic, versões
116 * - constant pool (via parse_constant_pool)
117 * - access flags, this_class, super_class
118 * - interfaces (aloca e preenche se necessário)
119 * - fields (aloca e chama read_member)
120 * - methods (aloca e chama read_member)
121 * - attributes_count (lê o valor; atributos de classe podem ser lidos por outra função)
122 *
123 * Observações importantes:
124 * - A função fecha o FILE passado (fclose) antes de retornar.
125 * - O chamador é responsável por liberar a memória alocada dentro do ClassFile
126 * (constant_pool, arrays, atributos) usando free_classfile().
127 *
128 * @param fptr Ponteiro para FILE já aberto (deve ser um .class).
129 * @return ClassFile Estrutura preenchida com os dados lidos. Campos não inicializados
130 * podem ocorrer se houver erro de leitura; é recomendado usar free_classfile
131 * após o uso.
132 */
133ClassFile read_classfile(FILE *fptr)
134{
135 ClassFile cf;
136
137 if (fseek(fptr, 0, SEEK_END) == 0)
138 {
139 const size_t fsize = ftell(fptr);
140
141 printf("Size: %lu bytes.\n", fsize);
142 fseek(fptr, 0, SEEK_SET);
143
144 cf.magic = read_u4(fptr);
145 cf.minor_version = read_u2(fptr);
146 cf.major_version = read_u2(fptr);
147 cf.constant_pool_count = read_u2(fptr);
148 cf.constant_pool = parse_constant_pool(fptr, cf.constant_pool_count);
149 cf.access_flags = read_u2(fptr);
150 cf.this_class = read_u2(fptr);
151 cf.super_class = read_u2(fptr);
152
153 /* Interfaces */
154 cf.interfaces_count = read_u2(fptr);
155 cf.interfaces = NULL;
156
157 if (cf.interfaces_count > 0)
158 {
159 cf.interfaces = (u2 *)calloc(cf.interfaces_count, sizeof(u2));
160 for (size_t i = 0; i < cf.interfaces_count; i++)
161 {
162 cf.interfaces[i] = read_u2(fptr);
163 }
164 }
165
166 /* Fields */
167 cf.fields_count = read_u2(fptr);
168 cf.fields = (member_info *)calloc(cf.fields_count, sizeof(member_info));
169 if (cf.fields != NULL)
170 read_member(cf.constant_pool, cf.fields_count, cf.fields, fptr);
171
172 /* Methods */
173 cf.methods_count = read_u2(fptr);
174 cf.methods = (member_info *)calloc(cf.methods_count, sizeof(member_info));
175 if (cf.methods != NULL)
176 read_member(cf.constant_pool, cf.methods_count, cf.methods, fptr);
177
178 cf.attributes_count = read_u2(fptr);
179 }
180
181 printf("%lu bytes readed.\n\n", ftell(fptr));
182 fclose(fptr);
183 return cf;
184}

186/**
187 * @brief Lê uma lista de member_info (fields ou methods) do arquivo e preenche o array.
188 *
189 * Para cada member, a função lê:
190 * - access_flags (u2)
191 * - name_index (u2)
192 * - descriptor_index (u2)
193 * - attributes_count (u2) e, se > 0, aloca e lê os atributos via read_attributes()
194 *
195 * @param cp Ponteiro para o constant pool (cp_info *) associado ao classfile.
196 * @param count Número de elementos (member_info) a serem lidos.
197 * @param info Ponteiro para o primeiro elemento de um array de member_info já alocado.
198 * @param fptr Ponteiro para o FILE de onde os dados serão lidos.
199 */
200void read_member(const cp_info *cp, u2 count, member_info *info, FILE *fptr)
201{
202 u2 attr_count;
203
204 for (size_t i = 0; i < count; i++)
205 {
206 info[i].access_flags = read_u2(fptr);
207 info[i].name_index = read_u2(fptr);
208 info[i].descriptor_index = read_u2(fptr);
209 info[i].attributes_count = attr_count = read_u2(fptr);
210
211 if (attr_count > 0)
212 {
213 info[i].attributes = (attribute *)calloc(attr_count, sizeof(attribute));
214 if (info[i].attributes != NULL)
215 read_attributes(cp, attr_count, fptr, info[i].attributes);
216 }
217 }
218}

220/**
221 * @brief Lê um array de atributos a partir do arquivo e preenche a estrutura `attribute`.
222 *
223 * Para cada atributo, a função:
224 * - lê attribute_name_index (u2) e attribute_length (u4)
225 * - obtém o nome do atributo a partir do constant pool e converte para o tipo de atributo
226 * (via convert_attr_name)
227 * - de acordo com o tipo (ex: ConstantValue, Code, LineNumberTable) faz leituras específicas
228 * e aloca buffers (code, exception_table, sub-attributes) conforme necessário.
229 *
230 * Observações:
231 * - Esta função acessa campos específicos do union `attribute.info` com base no tipo.
232 * - Caso o atributo possua sub-estruturas alocadas, estas ficam sob responsabilidade do
233 * chamador liberar posteriormente (por exemplo, através de free_attributes).
234 *
235 * @param cp Ponteiro para o constant pool (cp_info *) para resolução de nomes.
236 * @param n Número de atributos a serem lidos.
237 * @param fptr Ponteiro para o FILE de onde os atributos serão lidos.
238 * @param attr Ponteiro para o array de `attribute` previamente alocado que será preenchido.
239 */
240void read_attributes(const cp_info *cp, u2 n, FILE *fptr, attribute *attr)
241{
242 u2 c;
243
244 if (attr != NULL)
245 for (size_t i = 0; i < n; i++)
246 {
247 attr[i].attribute_name_index = read_u2(fptr);
248 attr[i].attribute_length = read_u4(fptr);
249
250 const wchar_t *attr_name = cp[attr[i].attribute_name_index - 1].info.UTF8.str;
251
252 const attribute_name *attr_type = convert_attr_name(attr_name);
253
254 if (attr_type != NULL)
255 switch (*attr_type)
256 {
257 case ConstantValue:
258 attr[i].info.ConstantValue.constantvalue_index = read_u2(fptr);
259 break;
260 case Code:
261 attr[i].info.Code.max_stack = read_u2(fptr);
262 attr[i].info.Code.max_locals = read_u2(fptr);
263 c = attr[i].info.Code.code_length = read_u4(fptr);
264 attr[i].info.Code.code = NULL;
265
266 if (c > 0)
267 {
268 attr[i].info.Code.code = (u1 *)malloc(c);
269 if (attr[i].info.Code.code != NULL)
270 fread(attr[i].info.Code.code, sizeof(u1), c, fptr);
271 }
272
273 c = attr[i].info.Code.exception_table_length = read_u2(fptr);
274 attr[i].info.Code.exception_table = NULL;
275
276 if (c > 0)
277 {
278 attr[i].info.Code.exception_table = (exception_table *)calloc(c, sizeof(exception_table));
279 if (attr[i].info.Code.exception_table != NULL)
280 for (size_t j = 0; j < c; j++)
281 {
282 attr[i].info.Code.exception_table[j].start_pc = read_u2(fptr);
283 attr[i].info.Code.exception_table[j].end_pc = read_u2(fptr);
284 attr[i].info.Code.exception_table[j].handler_pc = read_u2(fptr);
285 attr[i].info.Code.exception_table[j].catch_type = read_u2(fptr);
286 }
287 }
288
289 c = attr[i].info.Code.attributes_count = read_u2(fptr);
290 attr[i].info.Code.attributes = NULL;
291
292 if (c > 0)
293 {
294 attr[i].info.Code.attributes = (attribute *)calloc(c, sizeof(attribute));
295 if (attr[i].info.Code.attributes != NULL)
296 read_attributes(cp, c, fptr, attr[i].info.Code.attributes);
297 }
298 break;
299 case LineNumberTable:
300 c = attr[i].info.LineNumberTable.line_number_table_length = read_u2(fptr);
301 attr[i].info.LineNumberTable.line_number_table = NULL;
302
303 if (c > 0)
304 {
305 attr[i].info.LineNumberTable.line_number_table = (line_number_table *)calloc(c, sizeof(line_number_table));
306 if (attr[i].info.LineNumberTable.line_number_table != NULL)
307 for (size_t j = 0; j < c; j++)
308 {
309 attr[i].info.LineNumberTable.line_number_table[j].start_pc = read_u2(fptr);
310 attr[i].info.LineNumberTable.line_number_table[j].line_number = read_u2(fptr);
311 }
312 }
313 break;
314 default:

316 }
