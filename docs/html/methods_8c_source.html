57/**
58 * @brief Exibe todos os métodos definidos em um ClassFile.
59 *
60 * Para cada método, esta função:
61 * - Recupera nome e descritor no constant pool
62 * - Trata o método `<init>` (construtor), substituindo seu nome pelo nome da classe
63 * - Separa a lista de parâmetros e o tipo de retorno
64 * - Converte descritores JVM para texto legível via parse_descriptor()
65 * - Converte flags ACC_* em texto via parse_flags()
66 *
67 * Exemplo de saída:
68 *
69 * ```
70 * public void exemplo(int, float);
71 * descriptor: (IF)V
72 * flags: (0x0001) ACC_PUBLIC
73 * ```
74 *
75 * A função não altera o conteúdo original do `ClassFile` e libera toda memória
76 * auxiliar alocada dinamicamente durante o processo.
77 *
78 * @param cf Ponteiro para a estrutura ClassFile contendo os métodos.
79 */
80void show_methods(const ClassFile *cf)
81{
82 if (cf->constant_pool == NULL)
83 return;
84
85 cp_info *cp = cf->constant_pool;
86 u2 count = cf->methods_count;
87 char nl;
88
89 if (count > 0 && cf->methods != NULL)
90 for (size_t i = 0; i < count; i++)
91 {
92 /* Último método não imprime newline extra */
93 nl = i == (size_t)(count - 1) ? ' ' : '\n';
94
95 member_info method = cf->methods[i];
96 u2 access_flags = method.access_flags;
97
98 wchar_t *method_name = cp[method.name_index - 1].info.UTF8.str;
99 wchar_t *method_desc = cp[method.descriptor_index - 1].info.UTF8.str;
100
101 /* -------------------------------------------------------------------
102 * Tratamento especial para <init>
103 * -------------------------------------------------------------------
104 * Em Java, o construtor é mostrado como o nome da classe
105 * em vez de "<init>". Aqui fazemos essa substituição.
106 */
107 int is_init = !wcscmp(method_name, L"<init>");
108 if (is_init)
109 {
110 /* Recupera o nome da classe (this_class → CONSTANT_Class → Utf8) */
111 wchar_t *classname = cp[cp[cf->this_class - 1].info.Class.name_index - 1].info.UTF8.str;
112
113 /* Converte primeiro '/' encontrado em '.' */
114 for (wchar_t *wcptr = classname; wcptr < classname + wcslen(classname) - 1; wcptr++)
115 if (*wcptr == L'/')
116 {
117 *wcptr = L'.';
118 break;
119 }
120
121 method_name = classname;
122 }
123
124 /* -------------------------------------------------------------------
125 * Separação do descritor em parâmetros e tipo de retorno
126 * -------------------------------------------------------------------
127 */
128 wchar_t *params_end = wcschr(method_desc, L')') + 1;
129 size_t params_desc_len = params_end - method_desc;
130 size_t ret_desc_len = wcslen(params_end);
131
132 /* Copia substring dos parâmetros */
133 wchar_t *params_desc = (wchar_t *)calloc(params_desc_len + 1, sizeof(wchar_t));
134 if (params_desc)
135 {
136 wcsncpy(params_desc, method_desc, params_desc_len);
137 params_desc[params_desc_len] = L'\0';
138 }
139
140 /* Copia substring do tipo de retorno */
141 wchar_t *ret_desc = (wchar_t *)calloc(ret_desc_len + 1, sizeof(wchar_t));
142 if (ret_desc)
143 {
144 wcsncpy(ret_desc, params_end, ret_desc_len);
145 ret_desc[ret_desc_len] = L'\0';
146 }
147
148 /* Converte descritores */
149 wchar_t *params_str = parse_descriptor(params_desc, L",");
150 wchar_t *ret_str = NULL;
151 if (!is_init) /* Construtores não têm tipo de retorno */
152 ret_str = parse_descriptor(ret_desc, NULL);
153
154 /* Converte flags */
155 char *flags = parse_flags(access_flags, 12, ", ", flag_map);
156 char *kws = parse_flags(access_flags, 9, " ", flag_kw_map);
157
158 /* -------------------------------------------------------------------
159 * Impressão final
160 * -------------------------------------------------------------------
161 */
162 printf(" %s %ls%ls%ls;\n descriptor: %ls\n flags: (0x%04x) %s\n%c",
163 kws,
164 ret_str ? ret_str : L"", /* tipo de retorno */
165 method_name, /* nome do método */
166 params_str, /* lista de parâmetros */
167 method_desc, /* descritor bruto original */
168 access_flags,
169 flags,
170 nl);
171
172 /* Liberação de memória alocada dinamicamente */
173 free(flags);
174 free(kws);
175 free(params_desc);
176 free(ret_desc);
177 free(params_str);
178 free(ret_str);
179 }
180}
