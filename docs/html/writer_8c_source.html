4 *
5 * Este módulo contém a lógica responsável por exibir em stdout informações formatadas
6 * sobre um arquivo .class já carregado em memória. Ele mostra:

56/**
57 * @brief Exibe em stdout todas as informações relevantes de um ClassFile.
58 *
59 * A função exibe:
60 * - Palavra-chave correspondente às flags (ex.: "public final class X")
61 * - Magic number
62 * - Versão da classe
63 * - Índices e nomes de this_class e super_class
64 * - Contagens de interfaces, fields, methods e attributes
65 * - Todo o constant pool (via show_constants)
66 * - Fields e methods formatados (via show_fields e show_methods)
67 *
68 * O conteúdo exibido usa dados do constant pool para resolver nomes de classes,
69 * superclasses e UTF-8 strings armazenadas pela JVM.
70 *
71 * A função **não altera** a estrutura nem libera memória do ClassFile. A única
72 * memória liberada é a alocada internamente por parse_flags().
73 *
74 * @param cf Ponteiro para o ClassFile a ser exibido. Deve ser válido e carregado
75 * previamente por read_classfile().
76 */
77void show_classfile(ClassFile *cf)
78{
79 cp_info *cp = cf->constant_pool;
80
81 /* Recupera índices de nome da classe e superclasse */
82 u2 cls_name_index = cp[cf->this_class - 1].info.Class.name_index;
83 u2 super_cls_name_index = cp[cf->super_class - 1].info.Class.name_index;
84
85 /* Nomes das classes no constant pool (UTF-8 wide string) */
86 wchar_t *classname = cp[cls_name_index - 1].info.UTF8.str;
87 wchar_t *super_classname = cp[super_cls_name_index - 1].info.UTF8.str;
88
89 /* Conversão das flags para texto */
90 char *class_access_flags = parse_flags(cf->access_flags, 8, ", ", class_flag_map);
91 char *class_kws_flags = parse_flags(cf->access_flags, 6, " ", class_flag_kw_map);
92
93 /* Cabeçalho da classe */
94 printf("%s class %ls\n", class_kws_flags, classname);
95
96 printf(" Magic: %#X\n", cf->magic);
97 printf(" Version: %u.%u\n", cf->major_version, cf->minor_version);
98 printf(" Flags: (%#.4x) %s\n", cf->access_flags, class_access_flags);
99
100 printf(" this_class: #%u\t\t\t// %ls\n", cf->this_class, classname);
101 printf(" super_class: #%u\t\t\t// %ls\n", cf->super_class, super_classname);
102
103 printf(" interfaces: %u, fields: %u, methods: %u, attributes: %u\n",
104 cf->interfaces_count, cf->fields_count, cf->methods_count, cf->attributes_count);
105
106 /* Constant Pool */
107 printf("Constant Pool (count = %u):\n", cf->constant_pool_count);
108 if (cf->constant_pool_count > 0)
109 {
110 show_constants(cf->constant_pool_count, cp);
111 printf("{\n");
112 }
113
114 /* Fields */
115 if (cf->fields_count > 0)
116 {
117 show_fields(cf);
118 printf("\n");
119 }
120
121 /* Methods */
122 if (cf->methods_count > 0)
123 {
124 show_methods(cf);
125 printf("}\n");
126 }
127
128 /* Libera as strings alocadas por parse_flags */
129 free(class_access_flags);
130 free(class_kws_flags);
131}
