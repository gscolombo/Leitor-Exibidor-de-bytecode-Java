17/**
18 * @brief Inverte os dois bytes de um valor de 16 bits.
19 *
20 * Útil quando valores lidos do arquivo .class (big-endian) precisam ser tratados
21 * em arquitetura little-endian.
22 *
23 * @param n Valor de 16 bits.
24 * @return Valor com bytes trocados.
25 */
26unsigned int u2swap(const unsigned int n)
27{
28 return ((n >> 8) & 0xff) | /* Byte alto vai para baixo */
29 ((n << 8) & 0xff00); /* Byte baixo vai para cima */
30}

32/**
33 * @brief Inverte a ordem dos quatro bytes de um valor de 32 bits.
34 *
35 * Exemplo:
36 * - Entrada: 0x12345678
37 * - Saída: 0x78563412
38 *
39 * @param n Valor de 32 bits.
40 * @return Valor com bytes invertidos.
41 */
42unsigned int u4swap(const unsigned int n)
43{
44 return ((n >> 24) & 0xff) | /* Byte 3 → Byte 0 */
45 ((n >> 8) & 0xff00) | /* Byte 2 → Byte 1 */
46 ((n << 8) & 0xff0000) | /* Byte 1 → Byte 2 */
47 ((n << 24) & 0xff000000); /* Byte 0 → Byte 3 */
48}

54 * - n = 5 → 1

57 *
58 * @param n Inteiro não negativo.
59 * @return Quantidade de dígitos de `n`.
60 */
61unsigned int num_digits(unsigned int n)
62{
63 if (n < 10)
64 return 1;
65
66 int r = 2;
67 while ((n /= 10) > 9)
68 {
69 ++r;
70 }
71
72 return r;
73}

75/**
76 * @brief Converte flags da JVM (u2) em uma string legível.
77 *
78 * Dado um conjunto de bits representando flags (ex.: ACC_PUBLIC, ACC_STATIC),
79 * esta função combina os nomes das flags reconhecidas em um texto,
80 * separados pelo `sep` fornecido.
81 *
82 * A string retornada é alocada dinamicamente e deve ser liberada com `free()`.
83 *
84 * @param flags Valor de flags (u2).
85 * @param n Quantidade de itens no mapa de flags.
86 * @param sep Separador entre nomes das flags (ex.: " " ou ", ").
87 * @param flag_map Array de FlagMap com {bit,name}.
88 * @return char* String formatada contendo as flags ativas.
89 */
90char *parse_flags(u2 flags, size_t n, const char *sep, const FlagMap flag_map[])
91{
92 size_t sep_len = strlen(sep);
93 size_t buffer_size = 256;
94
95 char *flag_str = malloc(buffer_size);
96 if (flag_str)
97 {
98 char *ptr = flag_str;
99
100 for (size_t i = 0; i < n; i++)
101 {
102 if (flags & flag_map[i].flag)
103 {
104 size_t l = strlen(flag_map[i].name);
105
106 if (buffer_size > l + 2)
107 {
108 strcpy(ptr, flag_map[i].name);
109 ptr += l;
110 buffer_size -= l;
111
112 strcpy(ptr, sep);
113 ptr += sep_len;
114 buffer_size -= sep_len;
115 }
116 }
117 }
118
119 /* Remove separador extra ao final */
120 if (ptr > flag_str)
121 ptr -= sep_len;
122
123 *ptr = '\0';
124 }
125
126 /* Ajusta o buffer para o tamanho real usado */
127 char *tmp = realloc(flag_str, buffer_size);
128 if (tmp)
129 flag_str = tmp;
130
131 return flag_str;
132}

167/**
168 * @brief Converte um nome de atributo (wide-string) para o enum correspondente.
169 *
170 * Exemplo:
171 * - "Code" → attribute_name::Code
172 * - "LineNumberTable" → attribute_name::LineNumberTable
173 *
174 * @param name Nome do atributo em wchar_t (UTF-16/UTF-32 dependendo da plataforma).
175 * @return Ponteiro para o enum correspondente ou NULL se desconhecido.
176 */
177const attribute_name *convert_attr_name(const wchar_t *name)
178{
179 if (name != NULL)
180 for (size_t i = 0; i < sizeof(conversion) / sizeof(conversion[0]); i++)
181 {
182 if (!wcscmp(name, conversion[i].str))
183 return &conversion[i].attr;
184 }
185
186 return NULL;
187}

208/**
209 * @brief Converte descritores da JVM para representações legíveis.
210 *
211 * Exemplos:
212 * - "I" → "int"
213 * - "[I" → "int[]"
214 * - "Ljava/lang/String;" → "java.lang.String"
215 * - "(IF)V" → "( int float ) void"
216 *
217 * O parâmetro `sep` é inserido entre múltiplos tipos (por exemplo, em parâmetros
218 * de métodos).
219 *
220 * @param descriptor Descritor JVM em wchar_t.
221 * @param sep Wide-char separador (normalmente L"," ou L";").
222 * @return wchar_t* String alocada dinamicamente contendo o texto interpretado.
223 * Deve ser liberada com free().
224 */
225wchar_t *parse_descriptor(const wchar_t *descriptor, wchar_t *sep)
226{
227 if (descriptor == NULL)
228 return NULL;
229
230 unsigned int dim = 0;
231 size_t l = wcslen(descriptor);
232
233 wchar_t *buf = (wchar_t *)calloc(256, sizeof(wchar_t));
234
235 if (buf != NULL)
236 for (const wchar_t *desc = descriptor; desc < descriptor + l; desc++)
237 {
238 if (*desc == L'[')
239 {
240 dim++;
241 }
242 else
243 {
244 switch (*desc)
245 {
246 case L'(':
247 wcsncat(buf, L"(", 1);
248 break;
249
250 case L')':
251 wcsncat(buf, L")", 1);
252 break;
253
254 case L'L': /* Objeto: Lfully/qualified/ClassName; */
255 {
256 wchar_t *buf_it = buf + wcslen(buf);
257 while (*(++desc) != L';')
258 *(buf_it++) = (*desc == L'/') ? L'.' : *desc;
259
260 *buf_it = L'\0';
261 }
262 break;
263
264 default: /* Tipo primitivo */
265 for (size_t i = 0; i < sizeof(base_type_map) / sizeof(base_type_map[0]); i++)
266 if (*desc == base_type_map[i].base_type)
267 wcscat(buf, base_type_map[i].type);
268 }
269
270 /* Arrays */
271 while (dim > 0)
272 {
273 wcscat(buf, L"[]");
274 dim--;
275 }
276
277 /* Insere separador, se apropriado */
278 if (sep != NULL &&
279 desc > descriptor &&
280 desc + 1 < descriptor + l &&
281 *(desc + 1) != L')' &&
282 *desc != L'[' &&
283 *desc != L'V')
284 {
285 wcsncat(buf, sep, 1);
286 }
287
288 /* Insere espaço */
289 if (desc < descriptor + l &&
290 *desc != L'(' &&
291 *desc != L')' &&
292 *(desc + 1) != L')')
293 {
294 wcsncat(buf, L" ", 1);
295 }
296 }
297 }
298
299 return buf;
300}
