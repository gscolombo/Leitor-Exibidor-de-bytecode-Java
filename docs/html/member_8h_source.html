2 * @file include/types/member.h
3 * @brief Estruturas relacionadas a membros (fields e methods) do ClassFile.
4 *
5 * A especificação da JVM define a estrutura `member_info` para representar tanto
6 * campos (fields) quanto métodos (methods) de uma classe. Cada membro possui:
7 * - flags de acesso (public, static, final, ...)
8 * - índice para o nome (UTF-8) no constant pool
9 * - índice para o descritor no constant pool
10 * - lista de atributos associados
11 *
12 * Este header contém a estrutura `member_info` utilizada por fields e methods,
13 * além da estrutura `FlagMap`, usada para mapear flags binárias para strings
14 * legíveis nas funções de exibição.
15 */

40/**
41 * @struct member_info
42 * @brief Representa um campo (field) ou método (method) definido em um ClassFile.
43 *
44 * Cada `member_info` contém índices para o constant pool e uma lista de atributos
45 * específicos para o membro. A estrutura corresponde exatamente ao formato definido
46 * na Java Virtual Machine Specification.
47 *
48 * Campos:
49 * - **access_flags**: combinações de ACC_PUBLIC, ACC_STATIC, ACC_FINAL, etc.
50 * - **name_index**: índice no constant pool que aponta para uma entrada UTF8 com o nome.
51 * - **descriptor_index**: índice que aponta para o descritor JVM (ex.: "(I)V", "Ljava/lang/String;").
52 * - **attributes_count**: número de atributos do membro.
53 * - **attributes**: lista alocada dinamicamente de atributos (struct `attribute`).
54 */
55typedef struct member_info
56{
57 u2 access_flags; /**< Flags de acesso (ACC_*). */
58 u2 name_index; /**< Índice no constant pool para o nome UTF8 do membro. */
59 u2 descriptor_index; /**< Índice para o descritor do membro (parâmetros e retorno). */
60 u2 attributes_count; /**< Quantidade de atributos associados ao campo ou método. */
61 attribute *attributes;/**< Lista de atributos deste membro. */
62} member_info;

64#endif /* TYPES_INFO_H */
