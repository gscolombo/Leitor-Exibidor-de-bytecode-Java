69/**
70 * @brief Converte um conjunto de flags (u2) em uma string formatada.
71 *
72 * Flags de acesso da JVM (como ACC_PUBLIC, ACC_STATIC, ACC_FINAL) são representadas
73 * em bits dentro de um u2. Esta função converte esses bits em um texto legível,
74 * utilizando um mapa fornecido pelo chamador.
75 *
76 * Exemplo do formato retornado:
77 * ```
78 * "public static final"
79 * ```
80 *
81 * @param flags Valor u2 contendo bits correspondentes a flags da JVM.
82 * @param count Quantidade de entradas no array `map`.
83 * @param separator String usada para separar flags no texto final.
84 * @param map Array de mapeamento entre bits e nomes (FlagMap[], definido em member.h).
85 * @return char* String alocada dinamicamente contendo as flags convertidas.
86 * Deve ser liberada com free().
87 */
88char *parse_flags(u2 flags, size_t count, const char *separator, const FlagMap map[]);

111/**
112 * @brief Analisa um descritor JVM e retorna sua representação legível.
113 *
114 * Um descritor JVM pode representar:
115 * - tipos primitivos: `I`, `F`, `J`, `D`, `Z`...
116 * - objetos: `Ljava/lang/String;`
117 * - arrays: `[I`, `[[Ljava/lang/Object;`
118 * - descritores de métodos: `(ILjava/lang/String;)V`
119 *
120 * Esta função interpreta o descritor e escreve o resultado em forma textual.
121 *
122 * Exemplos:
123 * - Entrada: `"I"` → Saída: `"int"`
124 * - Entrada: `"[I"` → `"int[]"`
125 * - Entrada: `"(I)V"` → `"void method(int)"`
126 *
127 * @param desc Descritor de tipo da JVM (wide-string).
128 * @param out Buffer wide-string onde o resultado será armazenado.
129 * Deve ser previamente alocado pelo chamador.
130 * @return wchar_t* Ponteiro para `out`, contendo o texto convertido.
131 */
132wchar_t *parse_descriptor(const wchar_t *desc, wchar_t *out);
