3 * @brief Funções para exibição do Constant Pool (implementação).
4 *
5 * A função principal aqui é `show_constants` que percorre o array de `cp_info`
6 * e imprime cada entrada em formato legível, semelhante à saída do `javap`.
7 *
8 * Observações:
9 * - A função assume que as strings UTF-8 já foram decodificadas para `wchar_t*`
10 * (campo `UTF8.str`) durante o parsing do constant pool.
11 * - Alguns índices no constant pool referenciam outras entradas; o código faz
12 * ajustes com base no índice atual `i` para acessar corretamente as posições.
13 */

17/**
18 * @brief Exibe todas as entradas do constant pool.
19 *
20 * Para cada entrada, a função imprime uma linha no formato:
21 * #<index> = <TagName> <Detalhes>
22 *
23 * A função alinha os índices à direita calculando quantos espaços são
24 * necessários conforme o número máximo de entradas (`count`).
25 *
26 * @param count Número total de entradas no constant pool (u2).
27 * @param _cp Ponteiro para o array de cp_info (constant pool).
28 */
29void show_constants(u2 count, cp_info *_cp)
30{
31 /* i é o índice humano (começa em 1 conforme a especificação JVM) */
32 u2 i = 1;
33
34 /* Percorre o array de cp_info: cada cp aponta para a entrada atual */
35 for (cp_info *cp = _cp; cp < _cp + count; cp++, i++)
36 {
37 /*
38 * Alinhamento visual: calcula quantos espaços são necessários para alinhar
39 * os índices da lista à direita. Ex.: para 100 entradas, "# 1", "# 10", "#100".
40 */
41 u1 ws = num_digits(count) - num_digits(i);
42 char *pad = malloc(ws + 1);
43 if (pad == NULL)
44 {
45 /* Em caso de falha de alocação, garante não quebrar a execução */
46 pad = (char *)malloc(1);
47 if (pad == NULL)
48 return;
49 pad[0] = '\0';
50 }
51 memset(pad, ' ', ws);
52 pad[ws] = '\0';
53
54 switch (cp->tag)
55 {
56 case CONSTANT_Class:
57 {
58 /* A entrada Class armazena um índice para uma UTF8 que contém o nome. */
59 u2 cls_name_index = cp->info.Class.name_index - i;
60 /* Observação: impressão usa cp[cls_name_index].info.UTF8.str assumindo que
61 o ajuste por '- i' corresponde à organização do array no projeto. */
62 printf("%s#%u = Class\t\t\t#%u\t\t// %ls\n",
63 pad, i, cls_name_index, cp[cls_name_index].info.UTF8.str);
64 }
65 break;
66
67 case CONSTANT_Fieldref:
68 case CONSTANT_Methodref:
69 case CONSTANT_InterfaceMethodref:
70 {
71 /* Determina o rótulo textual (Fieldref / Methodref / InterfaceMethodref) */
72 char *ref;
73 if (cp->tag == CONSTANT_Fieldref)
74 ref = "Fieldref";
75 else if (cp->tag == CONSTANT_Methodref)
76 ref = "Methodref";
77 else
78 ref = "InterfaceMethodref";
79
80 /* Índices das estruturas de referência */
81 u2 cls_index = cp->info.Ref.class_index;
82 u2 name_and_type_index = cp->info.Ref.name_and_type_index;
83
84 /* Recupera as strings correspondentes (classe, nome e tipo) */
85 wchar_t *cls = cp[cp[cls_index - i].info.Class.name_index - i].info.UTF8.str;
86 wchar_t *ref_name = cp[cp[name_and_type_index - i].info.NameAndType.name_index - i].info.UTF8.str;
87 wchar_t *ref_type = cp[cp[name_and_type_index - i].info.NameAndType.descriptor_index - i].info.UTF8.str;
88
89 /*
90 * Para conformidade com a saída do javap, o método <init> é mostrado entre aspas.
91 * Substitui somente para exibição, sem alterar o conteúdo original.
92 */
93 ref_name = !wcscmp(ref_name, (wchar_t *)L"<init>") ? L"\"<init>\"" : ref_name;
94
95 printf("%s#%u = %s\t\t\t#%u.#%u\t\t// %ls.%ls:%ls\n",
96 pad, i, ref, cls_index, name_and_type_index, cls, ref_name, ref_type);
97 }
98 break;
99
100 case CONSTANT_String:
101 {
102 /* String referencia uma entrada UTF8 pelo índice string_index */
103 u2 str_index = cp->info.String.string_index;
104 printf("%s#%u = String\t\t\t#%u\t\t// %ls\n", pad, i, str_index, cp[str_index - i].info.UTF8.str);
105 }
106 break;
107
108 case CONSTANT_Integer:
109 printf("%s#%u = Integer\t\t\t%i\n", pad, i, cp->info._4Bn.number.i);
110 break;
111
112 case CONSTANT_Float:
113 printf("%s#%u = Float\t\t\t%f\n", pad, i, cp->info._4Bn.number.f);
114 break;
115
116 case CONSTANT_Long:
117 printf("%s#%u = Long\t\t\t%li\n", pad, i, cp->info._8Bn.number.l);
118 break;
119
120 case CONSTANT_Double:
121 printf("%s#%u = Double\t\t\t%f\n", pad, i, cp->info._8Bn.number.d);
122 break;
123
124 case CONSTANT_NameAndType:
125 {
126 /* Recupera índices e strings para NameAndType */
127 u2 name_index = cp->info.NameAndType.name_index;
128 u2 desc_index = cp->info.NameAndType.descriptor_index;
129
130 wchar_t *name = cp[name_index - i].info.UTF8.str;
131 wchar_t *desc = cp[desc_index - i].info.UTF8.str;
132
133 /* Mesma regra de exibição para <init> */
134 name = !wcscmp(name, (wchar_t *)L"<init>") ? L"\"<init>\"" : name;
135
136 printf("%s#%u = NameAndType\t\t#%u.#%u\t\t// %ls:%ls\n", pad, i, name_index, desc_index, name, desc);
137 }
138 break;
139
140 case CONSTANT_UTF8:
141 /* Strings UTF-8 já decodificadas em wchar_t* */
142 printf("%s#%u = UTF-8\t\t\t%ls\n", pad, i, cp->info.UTF8.str);
143 break;
144
145 case CONSTANT_MethodHandle:
146 printf("%s#%u = MethodHandle\n", pad, i);
147 break;
148
149 case CONSTANT_MethodType:
150 printf("%s#%u = MethodType\n", pad, i);
151 break;
152
153 case CONSTANT_InvokeDynamic:
154 printf("%s#%u = InvokeDynamic\n", pad, i);
155 break;
156
157 default:
158 /* Caso não reconhecido: não imprime nada específico */
159 break;
160 }
161
162 /* Libera o buffer de padding criado para o alinhamento */
163 free(pad);
164 }
165}
