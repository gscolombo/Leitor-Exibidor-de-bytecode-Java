52/**
53 * @brief Exibe todos os campos (fields) de um ClassFile.
54 *
55 * Para cada field, imprime algo como:
56 *
57 * ```
58 * public static int contador;
59 * descriptor: I
60 * flags: (0x0009) ACC_PUBLIC, ACC_STATIC
61 * ```
62 *
63 * Passos realizados:
64 * 1. Verifica se o constant pool existe.
65 * 2. Percorre `cf->fields_count`.
66 * 3. Resolve o nome do campo via constant pool.
67 * 4. Resolve o descritor JVM e converte para forma legível com parse_descriptor().
68 * 5. Converte flags de acesso em:
69 * - nomes completos (ACC_PUBLIC)
70 * - palavras-chave (public, static, ...)
71 * 6. Imprime a saída formatada.
72 *
73 * Observações:
74 * - `parse_descriptor()` e `parse_flags()` alocam memória, que é liberada aqui.
75 * - Esta função apenas exibe; não altera `cf` nem libera recursos internos dele.
76 *
77 * @param cf Ponteiro constante para a estrutura ClassFile contendo os fields.
78 */
79void show_fields(const ClassFile *cf)
80{
81 if (cf->constant_pool == NULL)
82 return;
83
84 cp_info *cp = cf->constant_pool;
85 u2 count = cf->fields_count;
86
87 char nl;
88
89 if (count > 0 && cf->fields != NULL)
90 for (size_t i = 0; i < count; i++)
91 {
92 /* Determina se há quebra de linha entre fields */
93 nl = i == (size_t)(count - 1) ? ' ' : '\n';
94
95 member_info field = cf->fields[i];
96
97 /* Índices para nome e descritor no constant pool */
98 u2 access_flags = field.access_flags;
99 wchar_t *field_name = cp[field.name_index - 1].info.UTF8.str;
100 wchar_t *field_desc = cp[field.descriptor_index - 1].info.UTF8.str;
101
102 /* Converte descritor JVM ("I", "Ljava/lang/String;") para forma legível ("int", "java.lang.String") */
103 wchar_t *field_desc_str = parse_descriptor(field_desc, NULL);
104
105 /* Converte flags para string completa (ACC_PUBLIC, ...) */
106 char *flags = parse_flags(access_flags, 9, ", ", flag_map);
107
108 /* Converte flags para palavras-chave Java (public, static, ...) */
109 char *kws = parse_flags(access_flags, 7, " ", flag_kw_map);
110
111 printf(" %s %ls%ls;\n descriptor: %ls\n flags: (0x%04x) %s\n%c",
112 kws, field_desc_str, field_name, field_desc,
113 access_flags, flags, nl);
114
115 /* Libera memória alocada por funções auxiliares */
116 free(flags);
117 free(field_desc_str);
118 free(kws);
119 }
120}
