21/**
22 * @struct ClassFile
23 * @brief Representa um arquivo .class Java carregado em memória.
24 *
25 * Campos principais:
26 * - magic: número mágico do arquivo (.class) (0xCAFEBABE).
27 * - minor_version / major_version: versão do formato da JVM.
28 * - constant_pool_count: tamanho do constant pool (1 + número de entradas).
29 * - constant_pool: array de entradas do constant pool (cp_info *).
30 * - access_flags: flags de acesso da classe (public, final, etc).
31 * - this_class: índice no constant pool para o nome desta classe.
32 * - super_class: índice no constant pool para o nome da superclasse.
33 * - interfaces_count / interfaces: número e lista de interfaces implementadas.
34 * - fields_count / fields: número e array de info de campos (member_info).
35 * - methods_count / methods: número e array de info de métodos (member_info).
36 * - attributes_count / attributes: número e array de atributos da classe.
37 */
38typedef struct ClassFile
39{
40 u4 magic;
41 u2 minor_version;
42 u2 major_version;
43 u2 constant_pool_count;
44 cp_info *constant_pool;
45 u2 access_flags;
46 u2 this_class;
47 u2 super_class;
48 u2 interfaces_count;
49 u2 *interfaces;
50 u2 fields_count;
51 member_info *fields;
52 u2 methods_count;
53 member_info *methods;
54 u2 attributes_count;
55 attribute *attributes;
56} ClassFile;
