23/**
24 * @brief Faz o parse completo do constant pool a partir do stream do .class.
25 *
26 * Lê `constant_pool_count` entradas do arquivo `f` (seguindo a especificação da JVM)
27 * e retorna um array alocado de `cp_info` com as entradas interpretadas.
28 *
29 * Observações:
30 * - A função aloca dinamicamente o array de cp_info e possivelmente buffers internos
31 * (por exemplo, para strings UTF8). É responsabilidade do chamador liberar
32 * essa memória (ex.: via free_classfile()).
33 * - A contagem `constant_pool_count` normalmente é lida do arquivo (.class) e
34 * passada para esta função diretamente.
35 *
36 * @param f Ponteiro para o FILE já posicionado após a leitura do constant_pool_count.
37 * @param constant_pool_count Número de entradas no constant pool (u2).
38 * @return cp_info* Ponteiro para o array alocado de cp_info. Retorna NULL em erro.
39 */
40cp_info* parse_constant_pool(FILE* f, u2 constant_pool_count);

42/**
43 * @brief Decodifica uma string em Modified UTF-8 para wchar_t*.
44 *
45 * A JVM armazena strings UTF-8 em um formato ligeiramente modificado (modified UTF-8).
46 * Esta função recebe o tamanho em bytes (u2) da sequência codificada e o buffer `data`
47 * contendo os bytes, e retorna uma string wide (`wchar_t*`) alocada dinamicamente
48 * com o conteúdo decodificado.
49 *
50 * Observações:
51 * - O buffer retornado deve ser liberado com `free()` pelo chamador.
52 * - Em caso de erro de alocação ou sequência inválida, a função pode retornar NULL.
53 *
54 * @param len Número de bytes do texto em modified UTF-8 (u2).
55 * @param data Ponteiro para os bytes codificados (u1*).
56 * @return wchar_t* String wide alocada contendo o texto decodificado, ou NULL em erro.
57 */
58wchar_t* decode_modified_utf8_str(u2 len, const u1* data);
59
60/**
61 * @brief Reconstrói um valor float a partir de 4 bytes (u4) seguindo IEEE 754.
62 *
63 * Interpreta `bytes` como a representação em 32 bits de um float (big-endian
64 * conforme leitura do .class) e retorna o `float` correspondente.
65 *
66 * @param bytes Valor u4 contendo os 4 bytes que representam o float.
67 * @return float Valor em ponto flutuante reconstruído.
68 */
69float decode_float_bytes(u4 bytes);

71/**
72 * @brief Reconstrói um valor long (64 bits) a partir de duas palavras de 32 bits.
73 *
74 * Recebe a parte alta (high_bytes) e a parte baixa (low_bytes) do long (ambas u4)
75 * e combina-as para retornar o `long` (64-bit).
76 *
77 * @param high_bytes Parte alta do valor (u4).
78 * @param low_bytes Parte baixa do valor (u4).
79 * @return long Valor long (64 bits) reconstruído.
80 */
81long decode_long_bytes(u4 high_bytes, u4 low_bytes);

83/**
84 * @brief Reconstrói um valor double (64 bits) a partir de duas palavras de 32 bits.
85 *
86 * Recebe a parte alta e baixa do double (u4 cada), combina-as em um valor de 64 bits
87 * e interpreta segundo o formato IEEE 754 para retornar o `double`.
88 *
89 * @param high_bytes Parte alta (u4).
90 * @param low_bytes Parte baixa (u4).
91 * @return double Valor de ponto flutuante dupla precisão reconstruído.
92 */
93double decode_double_bytes(u4 high_bytes, u4 low_bytes);
