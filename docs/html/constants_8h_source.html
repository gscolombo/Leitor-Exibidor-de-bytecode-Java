2 * @file include/types/cp/constants.h
3 * @brief Definições das constantes do Constant Pool e estruturas que representam cada tipo de entrada.
4 *
5 * O constant pool é uma das partes mais importantes do formato `.class`.
6 * Ele contém:
7 * - nomes de classes
8 * - tipos
9 * - literais numéricos
10 * - strings
11 * - descritores
12 * - referências a fields, methods e interfaces
13 * - metadados adicionais
14 *
15 * Este arquivo define os valores das tags (`CONSTANT_*`) usados no constant pool
16 * e as estruturas que armazenam cada variante de entrada, conforme a JVM Specification.

71/**
72 * @union Constant
73 * @brief Armazena o conteúdo real da entrada do constant pool dependendo da tag.
74 *
75 * Cada campo do union corresponde a um tipo de entrada permitido no constant pool:
76 * - CONSTANT_Class
77 * - CONSTANT_Fieldref / Methodref / InterfaceMethodref
78 * - CONSTANT_String
79 * - CONSTANT_Integer / Float
80 * - CONSTANT_Long / Double
81 * - CONSTANT_NameAndType
82 * - CONSTANT_UTF8
83 * - CONSTANT_MethodHandle
84 * - CONSTANT_MethodType
85 * - CONSTANT_InvokeDynamic
86 *
87 * Os campos seguem rigorosamente o formato especificado pela JVM.
88 */
89typedef union Constant
90{
91 /** @brief CONSTANT_Class: contém índice para o nome da classe (UTF8). */
92 struct
93 {
94 u2 name_index;
95 } Class;
96
97 /** @brief Fieldref / Methodref / InterfaceMethodref. */
98 struct
99 {
100 u2 class_index; /**< Índice para CONSTANT_Class */
101 u2 name_and_type_index; /**< Índice para CONSTANT_NameAndType */
102 } Ref;
103
104 /** @brief CONSTANT_String: índice para uma entrada UTF8. */
105 struct
106 {
107 u2 string_index;
108 } String;

111 struct
